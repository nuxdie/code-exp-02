cmake_minimum_required(VERSION 3.16)
project(white_window LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(X11 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TYPOGRAPHY REQUIRED IMPORTED_TARGET pangocairo fontconfig librsvg-2.0)
pkg_check_modules(GPU REQUIRED IMPORTED_TARGET epoxy)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
option(ENABLE_CLANG_TIDY "Enforce static analysis when compiling C++" ON)
find_program(CLANG_TIDY_EXECUTABLE NAMES clang-tidy-18 clang-tidy REQUIRED)
if(ENABLE_CLANG_TIDY)
    set(CMAKE_CXX_CLANG_TIDY "${CLANG_TIDY_EXECUTABLE};--quiet")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/.clang-tidy")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(CLANGD_EXECUTABLE NAMES clangd clangd-21 clangd-20 clangd-19 clangd-18)
if(NOT CLANGD_EXECUTABLE)
    set(CLANGD_EXECUTABLE clangd)
endif()
configure_file(viewer_config.h.in viewer_config.h @ONLY)
file(COPY fonts/noto-serif DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/fonts")
file(COPY assets DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")

foreach(FONT_FILE JetBrainsMono-Regular.ttf JetBrainsMono-Bold.ttf JetBrainsMono-Italic.ttf OFL.txt)
    configure_file("fonts/${FONT_FILE}" "fonts/${FONT_FILE}" COPYONLY)
endforeach()

# Embed each project source separately, including tests and build definitions.
file(GLOB_RECURSE VIEWER_SOURCES CONFIGURE_DEPENDS
    RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
    "*.cpp" "*.h" "*.hpp" "*.cc" "*.cxx" "*.c" "*.in" "*.cmake" "*.py" "*.yml" "CMakeLists.txt" ".clang-format" ".clang-tidy")
list(FILTER VIEWER_SOURCES EXCLUDE REGEX "(^|/)(build[^/]*|\\.git|\\.cache|fonts|assets)/")
list(REMOVE_ITEM VIEWER_SOURCES main.cpp camera.h canvas.h)
list(PREPEND VIEWER_SOURCES main.cpp camera.h canvas.h)

# Format before embedding the files, so the canvas displays the exact built code.
find_program(CLANG_FORMAT_EXECUTABLE NAMES clang-format-18 clang-format REQUIRED)
set(FORMAT_SOURCES "")
foreach(SOURCE_PATH IN LISTS VIEWER_SOURCES)
    if(SOURCE_PATH MATCHES "\\.(cpp|cc|cxx|c|h|hpp)$")
        list(APPEND FORMAT_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE_PATH}")
    endif()
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/.clang-format")
execute_process(COMMAND "${CLANG_FORMAT_EXECUTABLE}" --style=file -i ${FORMAT_SOURCES}
    RESULT_VARIABLE FORMAT_RESULT)
if(NOT FORMAT_RESULT EQUAL 0)
    message(FATAL_ERROR "Automatic C++ formatting failed")
endif()
add_custom_target(format
    COMMAND "${CLANG_FORMAT_EXECUTABLE}" --style=file -i ${FORMAT_SOURCES}
    COMMENT "Formatting C++ sources")
add_custom_target(format-check
    COMMAND "${CLANG_FORMAT_EXECUTABLE}" --style=file --dry-run --Werror ${FORMAT_SOURCES}
    COMMENT "Checking C++ formatting")

set(SOURCE_ENTRIES "")
foreach(SOURCE_PATH IN LISTS VIEWER_SOURCES)
    set(SOURCE_ABSOLUTE "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE_PATH}")
    string(FIND "${SOURCE_ABSOLUTE}" "${CMAKE_CURRENT_BINARY_DIR}/" IS_GENERATED)
    if(IS_GENERATED EQUAL 0)
        continue()
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE_ABSOLUTE}")
    file(READ "${SOURCE_ABSOLUTE}" SOURCE_TEXT)
    # A delimiter per file also lets the viewer display its own header template.
    string(SHA256 SOURCE_HASH "${SOURCE_TEXT}")
    string(SUBSTRING "${SOURCE_HASH}" 0 15 SOURCE_HASH)
    set(DELIMITER "S${SOURCE_HASH}")
    string(APPEND SOURCE_ENTRIES
        "    {\"${SOURCE_PATH}\", R\"${DELIMITER}(${SOURCE_TEXT})${DELIMITER}\"},\n")
endforeach()
configure_file(source_text.h.in source_text.h @ONLY)

add_library(language_server lsp.cpp)
target_compile_features(language_server PUBLIC cxx_std_17)
target_link_libraries(language_server PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_include_directories(language_server PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
find_program(LATEX_EXECUTABLE latex REQUIRED)
find_program(DVISVGM_EXECUTABLE dvisvgm REQUIRED)
add_library(math_typesetter math_typesetter.cpp)
target_compile_features(math_typesetter PUBLIC cxx_std_17)
target_include_directories(math_typesetter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(math_typesetter PUBLIC Threads::Threads)
find_program(NEATO_EXECUTABLE neato REQUIRED)
add_library(source_graph source_graph.cpp)
target_compile_features(source_graph PUBLIC cxx_std_17)
target_include_directories(source_graph PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(source_graph PUBLIC nlohmann_json::nlohmann_json)

add_library(gpu_renderer gpu.cpp)
target_compile_features(gpu_renderer PUBLIC cxx_std_17)
target_include_directories(gpu_renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gpu_renderer PUBLIC X11::X11 PkgConfig::TYPOGRAPHY PkgConfig::GPU)

add_library(application_core application.cpp application_model.cpp application_popups.cpp
    application_view.cpp application_events.cpp application_language.cpp application_math.cpp application_pointer.cpp popup_markup.cpp syntax.cpp source_panel.cpp)
target_compile_features(application_core PUBLIC cxx_std_17)
target_include_directories(application_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(application_core PUBLIC gpu_renderer language_server source_graph math_typesetter)

add_executable(white_window main.cpp)
target_compile_features(white_window PRIVATE cxx_std_17)
target_include_directories(white_window PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(white_window PRIVATE application_core)
